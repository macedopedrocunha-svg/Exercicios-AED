int maior_elemento(int v[], int n) {
    int maior = v[0];
    for (int i = 1; i < n; i++) {
        if (v[i] > maior) {
            maior = v[i];
        }
    }
    return maior;
}

- Complexidade de Tempo: O(n)
- Complexidade de Espaço Adicional: O(1)

void trocar_metades(int M[], int n) {
    int metade = n / 2;
    for (int i = 0; i < metade; i++) {
        int temp = M[i];
        M[i] = M[i + metade];
        M[i + metade] = temp;
    }
}

- Complexidade de Tempo: O(n)
- Complexidade de Espaço Adicional: O(1)


d) 
int* concatenar_vetores(int A[], int m, int B[], int n) {
    int* C = (int*) malloc((m + n) * sizeof(int));
    
    for (int i = 0; i < m; i++) {
        C[i] = A[i];
    }
    for (int j = 0; j < n; j++) {
        C[m + j] = B[j];
    }
    
    return C;
}

- Complexidade de Tempo: O(m + n)
- Complexidade de Espaço Adicional: O(m + n)  [devido à alocação do novo vetor C]

void imprimir_pares(int A[], int m, int B[], int n) {
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            printf("(%d, %d)\n", A[i], B[j]);
        }
    }
}

- Complexidade de Tempo: O(m * n)
- Complexidade de Espaço Adicional: O(1)

int somar_matriz(int n, int A[n][n]) {
    int soma = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            soma += A[i][j];
        }
    }
    return soma;
}

- Complexidade de Tempo: O(n²)
- Complexidade de Espaço Adicional: O(1)

int** matriz_transposta(int n, int A[n][n]) {
    int** T = (int**) malloc(n * sizeof(int*));
    for (int i = 0; i < n; i++) {
        T[i] = (int*) malloc(n * sizeof(int));
        for (int j = 0; j < n; j++) {
            T[i][j] = A[j][i];
        }
    }
    return T;
}

- Complexidade de Tempo: O(n²)
- Complexidade de Espaço Adicional: O(n²)  [devido à alocação da nova matriz T]


h) 
int** multiplicar_matrizes(int n, int A[n][n], int B[n][n]) {
    int** C = (int**) malloc(n * sizeof(int*));
    for (int i = 0; i < n; i++) {
        C[i] = (int*) malloc(n * sizeof(int));
        for (int j = 0; j < n; j++) {
            C[i][j] = 0;
            for (int k = 0; k < n; k++) {
                C[i][j] += A[i][k] * B[k][j];
            }
        }
    }
    return C;
}

- Complexidade de Tempo: O(n³)
- Complexidade de Espaço Adicional: O(n²)  [devido à alocação da nova matriz C]
